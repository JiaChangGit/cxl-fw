cmake_minimum_required(VERSION 3.20)
project(cxl_tierlab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TIERLAB_SANITIZERS "Enable Host address/undefined behavior sanitizers" OFF)
find_package(nlohmann_json 3.10 REQUIRED)
find_library(NUMA_LIBRARY numa REQUIRED)
add_library(tierlab_migrator src/migrator.cpp)
target_include_directories(tierlab_migrator PUBLIC src)
target_compile_options(tierlab_migrator PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(tierlab_policy src/policy.cpp src/workload.cpp)
target_link_libraries(tierlab_policy PUBLIC tierlab_migrator)
target_compile_options(tierlab_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(tierlab_runtime src/guest_runtime.cpp)
target_link_libraries(tierlab_runtime PUBLIC tierlab_migrator nlohmann_json::nlohmann_json ${NUMA_LIBRARY})
target_compile_options(tierlab_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(tierlab_drain src/drain.cpp)
target_link_libraries(tierlab_drain PUBLIC tierlab_migrator)
target_compile_options(tierlab_drain PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(tierlab src/main.cpp)
target_sources(tierlab PRIVATE src/policy_run.cpp src/drain_run.cpp src/trace_run.cpp)
target_link_libraries(tierlab PRIVATE tierlab_runtime tierlab_policy tierlab_drain)
target_compile_options(tierlab PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(TIERLAB_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(tierlab_migrator PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(tierlab_policy PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(tierlab_runtime PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(tierlab_drain PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(tierlab PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tierlab PRIVATE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(migrator_tests tests/cpp/migrator_tests.cpp)
  target_link_libraries(migrator_tests PRIVATE tierlab_migrator)
  target_compile_options(migrator_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME migrator_contract COMMAND migrator_tests)
  add_executable(mapping_tests tests/cpp/mapping_tests.cpp)
  target_link_libraries(mapping_tests PRIVATE tierlab_runtime)
  target_compile_options(mapping_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(mapping_tests PRIVATE -Wl,--wrap=mbind)
  add_test(NAME mapping_nodemask_contract COMMAND mapping_tests)
  add_executable(policy_tests tests/cpp/policy_tests.cpp)
  target_link_libraries(policy_tests PRIVATE tierlab_policy)
  target_compile_options(policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME policy_contract COMMAND policy_tests)
  add_executable(drain_tests tests/cpp/drain_tests.cpp)
  target_link_libraries(drain_tests PRIVATE tierlab_drain)
  target_compile_options(drain_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME drain_contract COMMAND drain_tests)
endif()
